cmake_minimum_required(VERSION 3.21)
project(narrator_engine_tests LANGUAGES CXX)
set(CMAKE_AUTOMOC ON)
set(CMAKE_CXX_STANDARD 17)
find_package(Qt6 6.10 REQUIRED COMPONENTS Core Gui)
add_executable(narrator_engine_tests main.cpp ../../internal/narratorservice.cpp ../../internal/narratorbackend.cpp ../../internal/narratorqueue.cpp)
target_include_directories(narrator_engine_tests PRIVATE ../..)
target_link_libraries(narrator_engine_tests PRIVATE Qt6::Core Qt6::Gui)
if(WIN32)
    target_compile_options(narrator_engine_tests PRIVATE /utf-8)
    target_link_libraries(narrator_engine_tests PRIVATE sapi ole32)
endif()
